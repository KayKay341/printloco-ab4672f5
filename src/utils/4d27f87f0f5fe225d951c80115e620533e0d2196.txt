import { type Unit } from "./units";

type AIContext = {
  serviceName?: string;
  fileName?: string;
  widthMm?: number;
  heightMm?: number;
  thicknessMm?: number;
  material?: string;
  quantity?: number;
  notes?: string;
  unit?: Unit;
};

let currentContext: AIContext = {};
const listeners = new Set<(ctx: AIContext) => void>();

export const aiContextStore = {
  set(ctx: AIContext) {
    currentContext = { ...currentContext, ...ctx };
    listeners.forEach((l) => l(currentContext));
  },
  get() {
    return currentContext;
  },
  subscribe(l: (ctx: AIContext) => void) {
    listeners.add(l);
    return () => listeners.delete(l);
  },
  clear() {
    currentContext = {};
    listeners.forEach((l) => l(currentContext));
  }
};
