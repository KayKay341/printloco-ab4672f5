export type Unit = "mm" | "in";

export const MM_PER_INCH = 25.4;

export function toMm(val: number, unit: Unit): number {
  if (unit === "mm") return val;
  return val * MM_PER_INCH;
}

export function fromMm(val: number, unit: Unit): number {
  if (unit === "mm") return val;
  return val / MM_PER_INCH;
}

export function formatUnit(val: number, unit: Unit): string {
  const precision = unit === "in" ? 3 : 1;
  return `${val.toFixed(precision)}${unit}`;
}

export function toInches(mm: number): number {
  return mm / MM_PER_INCH;
}

export function fromInches(inches: number): number {
  return inches * MM_PER_INCH;
}
